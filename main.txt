    <script>
        // Điều hướng màn hình
        function showDashboard() {
            document.getElementById('dashboard').classList.remove('hidden');
            document.getElementById('game-nim').classList.add('hidden');
            document.getElementById('btn-home').classList.add('hidden');
        }

        function startGame(gameId) {
            document.getElementById('dashboard').classList.add('hidden');
            document.getElementById('btn-home').classList.remove('hidden');
            if(gameId === 'nim') {
                document.getElementById('game-nim').classList.remove('hidden');
                document.getElementById('nim-setup').classList.remove('hidden');
                document.getElementById('nim-playboard').classList.add('hidden');
            }
        }

        // --- LOGIC TRÒ CHƠI BẪY SỐ THIÊN HÀ (NIM GAME) ---
        let targetN = 20;
        let maxK = 3;
        let currentNum = 0;
        let isPlayerTurn = true;
        let isGameOver = false;

        function initNimGame() {
            targetN = parseInt(document.getElementById('input-target').value) || 20;
            maxK = parseInt(document.getElementById('input-max-step').value) || 3;
            currentNum = 0;
            isGameOver = false;
            isPlayerTurn = true;

            document.getElementById('disp-target').innerText = targetN;
            document.getElementById('disp-max-step').innerText = maxK;
            document.getElementById('disp-current').innerText = currentNum;
            document.getElementById('nim-history').innerHTML = '<p class="text-slate-500 italic">Trò chơi bắt đầu từ 0...</p>';

            document.getElementById('nim-setup').classList.add('hidden');
            document.getElementById('nim-playboard').classList.remove('hidden');

            renderButtons();
            updateStatus("Lượt của bạn! Hãy chọn số cộng thêm.", "indigo");
        }

        function renderButtons() {
            const container = document.getElementById('action-buttons');
            container.innerHTML = '';
            
            for (let i = 1; i <= maxK; i++) {
                const btn = document.createElement('button');
                btn.className = "bg-slate-700 hover:bg-indigo-600 text-white font-bold py-3 rounded-xl transition text-lg disabled:opacity-50 disabled:cursor-not-allowed";
                btn.innerText = "+" + i;
                btn.onclick = () => handleMove(i);
                btn.disabled = !isPlayerTurn || (currentNum + i > targetN) || isGameOver;
                container.appendChild(btn);
            }
        }

        function handleMove(step) {
            if (isGameOver) return;

            currentNum += step;
            document.getElementById('disp-current').innerText = currentNum;
            
            const actor = isPlayerTurn ? "Bạn" : "Máy";
            addHistory(`${actor} đã cộng thêm +${step} (Tổng: ${currentNum})`);

            if (currentNum >= targetN) {
                isGameOver = true;
                if (isPlayerTurn) {
                    updateStatus("🎉 CHÚC MỪNG! Bạn đã chạm mốc " + targetN + " và CHIẾN THẮNG!", "emerald");
                } else {
                    updateStatus("💻 Máy đã chọn mốc " + targetN + ". MÁY THẮNG!", "rose");
                }
                renderButtons();
                return;
            }

            // Đổi lượt
            isPlayerTurn = !isPlayerTurn;
            renderButtons();

            if (!isPlayerTurn) {
                updateStatus("Máy đang suy nghĩ...", "amber");
                setTimeout(botMove, 1000);
            } else {
                updateStatus("Lượt của bạn!", "indigo");
            }
        }

        function botMove() {
            if (isGameOver) return;

            // Chiến thuật toán học: Tìm dư của (targetN - currentNum) chia cho (maxK + 1)
            let remainder = (targetN - currentNum) % (maxK + 1);
            let botStep = remainder;

            // Nếu nằm ở vị trí thua (remainder == 0), máy chọn ngẫu nhiên 1 bước
            if (remainder === 0 || remainder > maxK) {
                botStep = Math.floor(Math.random() * Math.min(maxK, targetN - currentNum)) + 1;
            }

            handleMove(botStep);
        }

        function updateStatus(text, color) {
            const statusEl = document.getElementById('disp-status');
            statusEl.innerText = text;
            statusEl.className = `py-2 px-4 rounded-xl bg-${color}-500/20 text-${color}-300 font-medium inline-block`;
        }

        function addHistory(text) {
            const historyEl = document.getElementById('nim-history');
            const p = document.createElement('p');
            p.innerText = text;
            historyEl.prepend(p);
        }
    </script>
